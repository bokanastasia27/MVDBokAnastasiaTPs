datos <- datos %>%
  mutate(
    # Estado civil
    marital_status = case_when(
      marital_status %in% c("Married-civ-spouse", "Married-AF-spouse") ~ "Casado",
      TRUE ~ "Soltero"
    ),
    # Educación
    education = case_when(
      education %in% c(
        "Preschool", "1st-4th", "5th-6th", "7th-8th",
        "9th", "10th", "11th", "12th"
      ) ~ "Educ. Primaria/intermedia",
      education == "HS-grad" ~ "Educ. Secundaria",
      education %in% c(
        "Some-college",
        "Assoc-acdm",
        "Assoc-voc"
      ) ~ "Educ. Terciaria",
      education == "Bachelors" ~ "Grado Completo",
      education %in% c(
        "Masters",
        "Prof-school",
        "Doctorate"
      ) ~ "Post Grad."
    ),
    # etnia
    race = case_when(
      race == "White" ~ "Blanco",
      TRUE ~ "No Blanco"     #al hacer un acercamiento a los datos se observan muy pocas personas no blancas, por lo tanto se agrupan las observaciones.    
    ),
    # Tipo de empleo
    workclass = case_when(
      workclass == "Private" ~ "Sector Privado",
      workclass %in% c(
        "Federal-gov",
        "Local-gov",
        "State-gov"
      ) ~ "Sector Publico",
      workclass %in% c(
        "Self-emp-inc",
        "Self-emp-not-inc"
      ) ~ "Independiente",
      
      TRUE ~ NA_character_
    )
    
  )

df_plot_or <- broom::tidy(modelo_2) %>%
  filter(term != "(Intercept)") %>%
  mutate(
    # Intervalos en escala log-odds
    lo_80 = estimate - qnorm(0.90) * std.error,
    hi_80 = estimate + qnorm(0.90) * std.error,
    lo_95 = estimate - qnorm(0.975) * std.error,
    hi_95 = estimate + qnorm(0.975) * std.error,
    lo_99 = estimate - qnorm(0.995) * std.error,
    hi_99 = estimate + qnorm(0.995) * std.error,

    # Pasamos todo a Odds Ratios
    across(
      c(estimate, starts_with("lo_"), starts_with("hi_")),
      exp
    ),

    # Etiquetas limpias
    term_clean = case_match(
      term,
      "sexMale" ~ "Sexo: Hombre vs Mujer",
      "age" ~ "Edad",
      "educationEduc. Primaria/intermedia" ~ "Educ. primaria/intermedia vs secundaria",
      "educationEduc. Terciaria" ~ "Educ. terciaria vs secundaria",

      "workclassIndependiente" ~ "Independiente vs privado",
      "marital_statusCasado" ~ "Casado vs soltero",
      "raceNo Blanco" ~ "No blanco vs blanco",
      "hours_per_week" ~ "Horas trabajadas por semana",
      "sexMale:workclassSector Publico" ~ "Interacción: Hombre × sector público",
      "sexMale:workclassIndependiente" ~ "Interacción: Hombre × independiente",

    )
  )

La comparación entre modelos muestra que la incorporación de la interacción entre sexo y tipo de empleo mejora el ajuste del modelo. El Modelo 2 presenta un menor valor de AIC (14124 frente a 14133) y un pseudo-R² ligeramente superior (0.3538 frente a 0.3533). Si bien la mejora en el pseudo-R² es moderada, la reducción del AIC sugiere que la mayor complejidad del modelo se encuentra justificada por una mejor capacidad explicativa. Por este motivo, el Modelo 2 es seleccionado como especificación final para el análisis. Asimismo, uno de los términos de interacción (Hombre × Independiente) resulta estadísticamente significativo, sugiriendo que el efecto del sexo sobre los ingresos varía según el tipo de empleo. En consecuencia, se considera que la inclusión de la interacción aporta información relevante y mejora el desempeño general del modelo, por lo que el Modelo 2 es seleccionado como especificación final.
